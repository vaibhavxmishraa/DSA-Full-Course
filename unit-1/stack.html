Algorithm 


PUSH(STACK, ITEM)

1. Check if stack is full.
2. If full, display OVERFLOW.
3. Otherwise increment TOP.
4. Insert ITEM at STACK[TOP].
5. Stop.

Algorithm 


POP(STACK)

1. Check if stack is empty.
2. If empty, display UNDERFLOW.
3. Otherwise store STACK[TOP].
4. Decrement TOP.
5. Return deleted element.
6. Stop.

Overflow & Underflow

🚨 Stack Overflow

Jab stack full ho aur hum new element push karne ki koshish karein. 


if(top == MAX - 1)
{
    cout << "Overflow";
}

⚠️ Stack Underflow

Jab stack empty ho aur hum pop karne ki koshish karein. 


if(top == -1)
{
    cout << "Underflow";
}

Stack using Array 


#include <iostream>

using namespace std;

#define MAX 5

int stack[MAX];

int top = -1;

void push(int value)
{
    if(top == MAX - 1)
    {
        cout << "Stack Overflow\n";
        return;
    }

    top++;

    stack[top] = value;
}

void pop()
{
    if(top == -1)
    {
        cout << "Stack Underflow\n";
        return;
    }

    cout << "Deleted: "
         << stack[top]
         << endl;

    top--;
}

void peek()
{
    if(top == -1)
    {
        cout << "Stack Empty\n";
        return;
    }

    cout << "Top: "
         << stack[top]
         << endl;
}

int main()
{
    push(10);
    push(20);
    push(30);

    peek();

    pop();

    peek();

    return 0;
}
