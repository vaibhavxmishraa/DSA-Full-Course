Beginning Deletion Logic

Step 1: Check whether list empty hai.
Step 2: Temporary pointer mein head store karo.
Step 3: Head ko next node par move karo.
Step 4: Old head ko delete karo.

End Deletion Logic

Step 1: Empty list check karo.
Step 2: Agar only one node hai, head = NULL.
Step 3: Second-last node tak traverse karo.
Step 4: Last node ko temporary pointer mein store karo.
Step 5: Second-last ka next NULL karo.
Step 6: Last node delete karo. 

Node* temp = head;

while(temp->next->next != NULL)
{
    temp = temp->next;
}

Node* last = temp->next;

temp->next = NULL;

delete last;

Pointer Bypassing

Deletion ka sabse important concept hai: target node ko bypass karna.

Target delete karne ke baad:

Node* temp = head;

for(int i = 1;

    i++)

}

Node* deleteNode =
    temp->next;

temp->next =
    deleteNode->next;

delete deleteNode;
⚠️ Golden Rule: Target node delete karne se pehle uske next node ka connection lose mat hone do.

Complete C++ Program 


#include <iostream>

using namespace std;


struct Node
{
    int data;

    Node* next;
};


void display(Node* head)
{
    Node* temp = head;

    while(temp != NULL)
    {
        cout << temp->data
             << " → ";

        temp = temp->next;
    }

    cout << "NULL\n";
}


void deleteBeginning(Node*& head)
{
    if(head == NULL)
        return;

    Node* temp = head;

    head = head->next;

    delete temp;
}


void deleteEnd(Node*& head)
{
    if(head == NULL)
        return;

    if(head->next == NULL)
    {
        delete head;

        head = NULL;

        return;
    }

    Node* temp = head;

    while(temp->next->next != NULL)
    {
        temp = temp->next;
    }

    Node* last = temp->next;

    temp->next = NULL;

    delete last;
}


void deletePosition(
    Node*& head,
    int position)
{
    if(head == NULL)
        return;


    if(position == 1)
    {
        deleteBeginning(head);

        return;
    }


    Node* temp = head;


    for(int i = 1;
        i < position - 1 &&
        temp != NULL;
        i++)
    {
        temp = temp->next;
    }


    if(temp == NULL ||
       temp->next == NULL)
    {
        cout
        << "Invalid Position\n";

        return;
    }


    Node* deleteNode =
        temp->next;


    temp->next =
        deleteNode->next;


    delete deleteNode;
}


int main()
{
    Node* head = NULL;


    // Creating list

    Node* a =
        new Node{10, NULL};

    Node* b =
        new Node{20, NULL};

    Node* c =
        new Node{30, NULL};

    Node* d =
        new Node{40, NULL};


    head = a;

    a->next = b;

    b->next = c;

    c->next = d;


    cout
    << "Original List:\n";

    display(head);


    deleteBeginning(head);

    cout
    << "After Beginning Deletion:\n";

    display(head);


    deleteEnd(head);

    cout
    << "After End Deletion:\n";

    display(head);


    deletePosition(head, 2);

    cout
    << "After Position Deletion:\n";

    display(head);


    return 0;
}

Time Complexity

Deletion Time Reason
Beginning O(1) Direct head access
End O(n) Second-last node find karna
Specific Position O(n) Position tak traversal

Important Edge Cases

Empty List


head == NULL

Deletion perform nahi karni.

Only One Node


10 → NULL

head = NULL

Node delete hone ke baad list empty ho jayegi.

Position = 1

Specific-position function mein position 1 ko beginning deletion ki tarah handle karo.

Invalid Position

Agar requested position exist nahi karti, node delete nahi karna chahiye.
